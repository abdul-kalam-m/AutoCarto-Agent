\section*{Figures}

\begin{figure}[htbp]\centering
\includegraphics[width=\textwidth]{figures/ungated_vs_gated.png}
\caption{\textbf{Identical data, identical geometry, opposite outcomes.} Left: an
unconstrained proposal collapses 414 of 530 tracts (78\%) into one class. Right:
Gate 2 prescribes a log transform plus Jenks natural breaks, yielding balanced
classes [98/134/145/90/63]. The rejected classification attains a \emph{higher}
goodness-of-variance fit (0.866 vs 0.835); class balance, not GVF, is the honest
failure metric.}
\end{figure}

\begin{figure}[htbp]\centering
\includegraphics[width=\textwidth]{figures/architecture_boundary.png}
\caption{\textbf{Three-tier architecture and the authority boundary.} Raw data
values never cross into Tier 1, and no numeric constant reaches the executed
render without deterministic provenance.}
\end{figure}

\begin{figure}[htbp]\centering
\includegraphics[width=\textwidth]{figures/gate2_distribution_diagnostics.png}
\caption{\textbf{Gate 2 distribution diagnostics across five regimes,} with
proposed and prescribed break positions overlaid.}
\end{figure}

\begin{figure}[htbp]\centering
\includegraphics[width=\textwidth]{figures/gate3b_bivariate_scenarios.png}
\caption{\textbf{Gate 3b decision triptych:} strong coupling, weak coupling, and
independence, yielding APPROVE, WARN, and REJECT.}
\end{figure}

\begin{figure}[htbp]\centering
\includegraphics[width=\textwidth]{figures/atlanta_results_panel.png}
\caption{\textbf{Controlled Atlanta case.} Bivariate encoding unlocked only after
Gate 3b confirmed cross-correlation ($I_{xy}=+0.3262$, pseudo-$p=0.0050$, 199
permutations; Spearman $\rho=+0.9471$).}
\end{figure}
